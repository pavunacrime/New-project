  <script>
    const canvas = document.getElementById('rainCanvas');
    const ctx = canvas.getContext('2d');

    function resizeCanvas() {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    }
    resizeCanvas();
    window.addEventListener('resize', resizeCanvas);

    // Emojis de armas para a chuva
    const weapons = ['🔫', '🗡️', '⚔️', '💣', '🔪', '🛡️'];
    const fontSize = 22;
    const columns = Math.floor(canvas.width / (fontSize + 10));
    
    // Array com as posições Y de cada coluna
    const drops = [];
    for (let i = 0; i < columns; i++) {
      drops[i] = Math.random() * -100; // Início assíncrono topo
    }

    function draw() {
      // Limpa com leve opacidade para criar rastro
      ctx.fillStyle = 'rgba(0, 0, 0, 0.15)';
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      ctx.font = `${fontSize}px sans-serif`;

      for (let i = 0; i < drops.length; i++) {
        // Seleciona uma arma aleatória
        const text = weapons[Math.floor(Math.random() * weapons.length)];
        const x = i * (fontSize + 10);
        const y = drops[i] * fontSize;

        // Desenha o emoji da arma
        ctx.fillText(text, x, y);

        // Se a arma chegou ao final da tela, reseta aleatoriamente para o topo
        if (y > canvas.height && Math.random() > 0.975) {
          drops[i] = 0;
        }

        drops[i]++;
      }
    }

    // Roda a animação a 30 FPS
    setInterval(draw, 33);
  </script>
